# Allod appliance: Raspberry Pi 4 (kernel8.img) and Pi 5 (kernel_2712.img).
# The firmware picks the matching initramfs (initramfs8 / initramfs_2712).
[all]
arm_64bit=1
auto_initramfs=1
# Fast boot: no rainbow screen, no boot delay.
disable_splash=1
boot_delay=0
# KMS display driver, used by the splash and the kiosk.
dtoverlay=vc4-kms-v3d
max_framebuffers=2
disable_overscan=1
# USB boot and USB SSDs: full current on the USB ports (Pi 5 with a 5 A supply).
usb_max_current_enable=1

[pi4]
arm_boost=1

[pi5]
# NVMe on the PCIe connector: gen 3 is faster and works with most drives.
dtparam=pciex1
dtparam=pciex1_gen=3
